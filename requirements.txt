# ============================================================================

# 版本策略：这里每一条都钉死到具体版本，而不是写 >= 让 pip 自己挑。

# ---- Web 层 ----------------------------------------------------------------
fastapi==0.141.1
uvicorn[standard]==0.54.0

# ---- 数据层 ----------------------------------------------------------------
sqlalchemy==2.0.54
# ↑ 2.1 已经发布，但网上教程和 AI 生成的代码全是 2.0 的写法
#   （DeclarativeBase / Mapped[] / mapped_column）。
#   为一个作品集项目去趟 2.1 的浑水不划算，等社区写法跟上再说。

# ---- 采集层 ----------------------------------------------------------------
httpx==0.28.1
# ↑ 用它而不是 requests：采集层要跑在 async worker 里，
#   requests 是同步的，会把事件循环堵死。

# ---- 数据处理 --------------------------------------------------------------
numpy==2.5.2
# ↑ ★ 自己直接 import 的库必须自己钉版本。
#   seed.py 用 numpy 做对数正态/幂律采样。如果靠 pandas 传递安装，
#   哪天 pandas 改了依赖声明（比如换成 numpy>=3 或者干脆换成别的数组库），
#   seed.py 会立刻 import 失败，而报错现场在数据生成，不在依赖清单——
#   排查成本远高于写这一行的成本。

pandas==3.0.3
# ↑ 只用于 export（CSV/Excel）。清洗层刻意不用它，原因见 docs/可行性评估.md：
#   pandas 3.0 的字符串 dtype 从 object 变成 str、None 被强制转 NaN、
#   字符串列塞 float 直接 TypeError——网上教程全是 2.x 写法，
#   而我们的清洗场景是逐条字符串解析，向量化的优势一点都用不上。

openpyxl==3.1.5
# ↑ pandas.to_excel 的引擎，不显式依赖的话 Excel 导出会在运行时才报错。

matplotlib==3.11.2
# ↑ ★ 它不是"画演示图的可选工具"，而是**开发期自检工具**：
#   spike/plot_seed.py 画的三张图是发现数据 bug 的主要手段
#   （价格累乘漂移、大促日被跳过，两个 bug 都是图上先看出来的，
#     统计指标当时全是"正常"的——详见 tests/test_seed.py 顶部注释）。
#   因此在 requirements.txt 里是硬依赖，不是 dev 依赖。
#
#   两个使用上的坑（spike/plot_seed.py 里都处理了）：
#     ① 必须在 import pyplot **之前** matplotlib.use("Agg")，
#        否则无头环境（CI、远程终端）会挂；
#     ② 必须设置中文字体，否则所有中文标签渲染成一排方框（tofu），
#        图上写着"□□□□"，等于没画。

# ---- 配置 / 中文处理 -------------------------------------------------------
python-dotenv==1.2.3

jieba==0.42.1
# ↑ 中文分词，词云用。

setuptools<81
# ↑ ★ 这行看着莫名其妙，但删掉项目就跑不起来。
#   jieba 的 _compat.py 至今仍写着 `import pkg_resources`，
#   而 setuptools 82（2026-02 发布）已经把 pkg_resources 整个删除了。
#   症状是：阶段⑤ 做词云时直接 ModuleNotFoundError: No module named 'pkg_resources'，
#   而你会先去怀疑 jieba 是不是装坏了——实际是 setuptools 太新。

# ---- 定时任务 --------------------------------------------------------------
apscheduler==3.11.3
# ↑ ★ 不要升 4.x。4.x 至今只有 4.0.0a6 这种 alpha 版本，
#   官方 README 里明写着「请勿用于生产」。

# ---- 采集降级层（可选）-----------------------------------------------------
playwright>=1.48
# ↑ ★ 注意：装包 ≠ 装浏览器。
#   `pip install playwright` 只装 Python 库（几 MB），
#   真正跑起来还需要 `playwright install chromium` 下载约 150MB 的浏览器二进制。
#   本项目的 SMZDM 页面是服务端渲染、httpx 就够用，Playwright 只是
#   分层降级链的最后一层。所以：**现在不要执行 install chromium**，
#   等阶段④ 真的遇到需要 JS 渲染的页面时再说。
#   这里用 >= 而非 == ，是因为 playwright 对 Python 版本有要求
#   （3.13 需要 >=1.48），允许它往上走一点比钉死更稳妥。

# ---- 测试 ------------------------------------------------------------------
pytest==9.1.1
